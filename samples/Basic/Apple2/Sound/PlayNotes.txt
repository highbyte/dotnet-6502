10 REM * APPLE II SPEAKER - PLAY NOTES *
20 REM
30 REM APPLESOFT HAS NO SOUND COMMAND.
40 REM ALL SOUND ON AN APPLE II IS MADE
50 REM BY TOGGLING THE SPEAKER AT $C030,
60 REM AND THE PITCH IS SIMPLY HOW OFTEN
70 REM YOU DO IT. BASIC IS FAR TOO SLOW
80 REM TO REACH MUSICAL RATES, SO THIS
90 REM POKES A SHORT MACHINE CODE TONE
100 REM ROUTINE INTO PAGE 3 AND CALLS IT
110 REM ONCE PER NOTE.
120 REM
130 REM $317 = PITCH  DELAY BETWEEN
140 REM        TOGGLES. BIGGER IS LOWER.
150 REM $318 = LENGTH HOW MANY TOGGLES.
160 REM
170 HOME
180 PRINT "APPLE II SPEAKER DEMO"
190 PRINT
200 DIM P(8),L(8),A(4)
210 PRINT "POKING TONE ROUTINE AT $300..."
220 FOR I = 0 TO 22: READ B: POKE 768 + I,B: NEXT I
230 DATA 173,24,3,141,25,3,173,48,192,172
240 DATA 23,3,234,234,136,208,251,206,25,3
250 DATA 208,240,96
260 REM
270 REM C MAJOR SCALE, C4 UP TO C5
280 FOR N = 1 TO 8: READ P(N),L(N): NEXT N
290 DATA 215,131,191,147,170,165,161,175
300 DATA 143,196,127,220,113,247,107,255
310 REM
320 REM WHICH OF THOSE MAKE THE CHORD
330 FOR S = 1 TO 4: READ A(S): NEXT S
340 DATA 1,3,5,8
350 REM
360 PRINT "SCALE UP..."
370 FOR N = 1 TO 8: GOSUB 900: NEXT N
380 PRINT "SCALE DOWN..."
390 FOR N = 8 TO 1 STEP -1: GOSUB 900: NEXT N
400 PRINT "ARPEGGIO..."
410 FOR R = 1 TO 3
420 FOR S = 1 TO 4: N = A(S): GOSUB 900: NEXT S
430 NEXT R
440 PRINT
450 PRINT "DONE."
460 END

900 REM * PLAY NOTE N *

920 CALL 768
